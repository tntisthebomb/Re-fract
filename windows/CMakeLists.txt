cmake_minimum_required(VERSION 3.21)
# Statically link the MSVC runtime so Re-fract.exe needs no redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(ReFract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

# SDL2: use an installed copy if there is one, otherwise build 2.30.9 statically.
find_package(SDL2 QUIET)
if(NOT SDL2_FOUND)
  include(FetchContent)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON CACHE BOOL "" FORCE)
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL2 URL https://github.com/libsdl-org/SDL/archive/refs/tags/release-2.30.9.tar.gz)
  FetchContent_MakeAvailable(SDL2)
endif()
if(TARGET SDL2::SDL2)
  set(RF_SDL SDL2::SDL2)
elseif(TARGET SDL2::SDL2-static)
  set(RF_SDL SDL2::SDL2-static)
elseif(TARGET SDL2-static)
  set(RF_SDL SDL2-static)
else()
  set(RF_SDL ${SDL2_LIBRARIES})
endif()
find_package(Threads REQUIRED)
find_package(OpenGL REQUIRED)

set(RF_SOURCES src/main.cpp src/gl_loader.cpp
  engine/expression.cpp engine/formula.cpp engine/renderer.cpp engine/storage.cpp engine/surface.cpp engine/ui.cpp)
# The GPU ray-marcher is compiled in when present; otherwise the CPU backend is the only one.
set(RF_GPU_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/src/gpu_renderer.cpp" CACHE FILEPATH "GPU renderer source")
if(EXISTS "${RF_GPU_SOURCE}")
  list(APPEND RF_SOURCES "${RF_GPU_SOURCE}")
  set(RF_HAVE_GPU ON)
endif()

add_executable(Re-fract WIN32 ${RF_SOURCES})
target_include_directories(Re-fract PRIVATE src engine)
target_link_libraries(Re-fract PRIVATE ${RF_SDL} Threads::Threads OpenGL::GL)
if(TARGET SDL2::SDL2main)
  target_link_libraries(Re-fract PRIVATE SDL2::SDL2main)
elseif(TARGET SDL2main)
  target_link_libraries(Re-fract PRIVATE SDL2main)
endif()
if(RF_HAVE_GPU)
  target_compile_definitions(Re-fract PRIVATE RF_HAVE_GPU=1)
  get_filename_component(RF_GPU_DIR "${RF_GPU_SOURCE}" DIRECTORY)
  target_include_directories(Re-fract PRIVATE "${RF_GPU_DIR}")
endif()
if(MSVC)
  target_compile_options(Re-fract PRIVATE /W3 /utf-8)
  target_compile_definitions(Re-fract PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(Re-fract PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
# Copy SDL2.dll next to the exe when SDL2 was found as a shared library.
add_custom_command(TARGET Re-fract POST_BUILD COMMAND ${CMAKE_COMMAND} -E
  $<IF:$<BOOL:$<TARGET_RUNTIME_DLLS:Re-fract>>,copy,true> $<TARGET_RUNTIME_DLLS:Re-fract> $<TARGET_FILE_DIR:Re-fract>
  COMMAND_EXPAND_LISTS)
